cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(transceiver_bringup C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_BOARD STREQUAL "pico")
  message(FATAL_ERROR "This bring-up image requires the original Pico")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" FW_VERSION)
string(REPLACE "." ";" FW_PARTS "${FW_VERSION}")
list(GET FW_PARTS 0 FW_MAJOR)
list(GET FW_PARTS 1 FW_MINOR)
list(GET FW_PARTS 2 FW_PATCH)
# Exported build directories have no reliable Git context: require exact inputs.
string(LENGTH "${EMBER_GIT_REVISION}" REV_LENGTH)
if(NOT REV_LENGTH EQUAL 40 OR NOT EMBER_GIT_REVISION MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Supply -DEMBER_GIT_REVISION=<full source commit>")
endif()
if(NOT "${EMBER_GIT_DIRTY}" MATCHES "^[01]$")
  message(FATAL_ERROR "Supply -DEMBER_GIT_DIRTY=0 or 1")
endif()
configure_file(build_identity.h.in build_identity.h @ONLY)
add_executable(transceiver_bringup main.c)
target_include_directories(transceiver_bringup PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
pico_set_program_version(transceiver_bringup "${FW_VERSION}")
target_include_directories(transceiver_bringup PRIVATE ../comms_transport)
target_compile_options(transceiver_bringup PRIVATE -Wall -Wextra -Werror)
target_link_libraries(transceiver_bringup pico_stdlib hardware_i2c hardware_adc hardware_spi pico_rand)
pico_enable_stdio_usb(transceiver_bringup 1)
pico_enable_stdio_uart(transceiver_bringup 0)
pico_add_extra_outputs(transceiver_bringup)
