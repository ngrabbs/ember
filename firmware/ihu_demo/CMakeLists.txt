cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
set(PICO_BOARD pico CACHE STRING "Demo supports the original non-wireless RP2040 Pico")
if(NOT PICO_BOARD STREQUAL "pico")
    message(FATAL_ERROR "ihu_demo requires PICO_BOARD=pico (original RP2040 Pico/Pico H)")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ihu_demo C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" FW_VERSION)
if(NOT EMBER_GIT_REVISION MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Supply source identity with firmware/transceiver/source_identity.py")
endif()
string(LENGTH "${EMBER_GIT_REVISION}" REV_LENGTH)
if(NOT REV_LENGTH EQUAL 40 OR NOT "${EMBER_GIT_DIRTY}" MATCHES "^[01]$")
    message(FATAL_ERROR "Full 40-character source revision and dirty=0|1 required")
endif()
include(FetchContent)
FetchContent_Declare(ember_freertos
    URL https://codeload.github.com/FreeRTOS/FreeRTOS-Kernel/tar.gz/0adc196d4bd52a2d91102b525b0aafc1e14a2386
    URL_HASH SHA256=6af5d78e39a5237b63faaafdc31672c87b27b68e713e2a57d3528aa3a4c1cee8)
FetchContent_GetProperties(ember_freertos)
if(NOT ember_freertos_POPULATED)
    FetchContent_Populate(ember_freertos)
endif()
set(FREERTOS_KERNEL_PATH ${ember_freertos_SOURCE_DIR})
set(FREERTOS_CONFIG_FILE_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
list(APPEND CMAKE_MODULE_PATH ${PICO_SDK_PATH}/external)
add_subdirectory(${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2040 freertos)
add_executable(ihu_demo main.c demo.c)
target_compile_options(ihu_demo PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(ihu_demo PRIVATE
    DEMO_VERSION="${FW_VERSION}" DEMO_GIT="${EMBER_GIT_REVISION}" DEMO_DIRTY=${EMBER_GIT_DIRTY}
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0
    PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0)
target_link_libraries(ihu_demo pico_stdlib hardware_watchdog FreeRTOS-Kernel-Static)
pico_enable_stdio_usb(ihu_demo 1)
pico_enable_stdio_uart(ihu_demo 0)
pico_set_program_name(ihu_demo "EMBER IHU DEMO ONLY")
pico_set_program_version(ihu_demo "${FW_VERSION}")
pico_add_extra_outputs(ihu_demo)
